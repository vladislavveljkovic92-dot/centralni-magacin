# 1. BAZA PODATAKA (Autopatch)
# ==========================================
def init_db():
    conn = sqlite3.connect(DB_NAME)
    c = conn.cursor()
    
    # Osnovne tabele
    c.execute('''CREATE TABLE IF NOT EXISTS delovi (id INTEGER PRIMARY KEY AUTOINCREMENT, part_number TEXT UNIQUE, naziv TEXT, namena TEXT DEFAULT 'Univerzalno', kategorija TEXT, stanje INTEGER, min_stanje INTEGER, max_stanje INTEGER, cena REAL, dobavljac TEXT, slika_putanja TEXT DEFAULT '')''')
    c.execute('''CREATE TABLE IF NOT EXISTS transakcije (id INTEGER PRIMARY KEY AUTOINCREMENT, part_id INTEGER, kolicina INTEGER, tip TEXT, korisnik TEXT, datum DATETIME, napomena TEXT, FOREIGN KEY(part_id) REFERENCES delovi(id))''')
    c.execute('''CREATE TABLE IF NOT EXISTS podesavanja (kljuc TEXT PRIMARY KEY, vrednost TEXT)''')
    c.execute('''CREATE TABLE IF NOT EXISTS korisnici (id INTEGER PRIMARY KEY AUTOINCREMENT, username TEXT UNIQUE, password TEXT, uloga TEXT)''')
    c.execute('''CREATE TABLE IF NOT EXISTS radni_nalozi (id INTEGER PRIMARY KEY AUTOINCREMENT, rn_broj TEXT UNIQUE, status TEXT, datum DATETIME, napomena TEXT, korisnik TEXT)''')
    c.execute('''CREATE TABLE IF NOT EXISTS rn_stavke (id INTEGER PRIMARY KEY AUTOINCREMENT, rn_id INTEGER, part_id INTEGER, kolicina INTEGER, FOREIGN KEY(rn_id) REFERENCES radni_nalozi(id), FOREIGN KEY(part_id) REFERENCES delovi(id))''')
    
    # PM Tabele
    c.execute('''CREATE TABLE IF NOT EXISTS pm_linije (id INTEGER PRIMARY KEY AUTOINCREMENT, naziv TEXT UNIQUE, opis TEXT, ikona TEXT DEFAULT '🏭')''')
    c.execute('''CREATE TABLE IF NOT EXISTS pm_masine (id INTEGER PRIMARY KEY AUTOINCREMENT, linija_id INTEGER, naziv TEXT, FOREIGN KEY(linija_id) REFERENCES pm_linije(id))''')
    c.execute('''CREATE TABLE IF NOT EXISTS pm_zadaci (id INTEGER PRIMARY KEY AUTOINCREMENT, masina_id INTEGER, opis_zadatka TEXT, tip_odrzavanja TEXT, FOREIGN KEY(masina_id) REFERENCES pm_masine(id))''')
    c.execute('''CREATE TABLE IF NOT EXISTS pm_evidencija (id INTEGER PRIMARY KEY AUTOINCREMENT, zadatak_id INTEGER, status TEXT, korisnik TEXT, datum DATETIME, napomena TEXT, FOREIGN KEY(zadatak_id) REFERENCES pm_zadaci(id))''')

    # Budzet Tabele
    c.execute('''CREATE TABLE IF NOT EXISTS budzet_odobren (id INTEGER PRIMARY KEY AUTOINCREMENT, godina INTEGER, kategorija TEXT UNIQUE, odobrena_suma REAL)''')
    c.execute('''CREATE TABLE IF NOT EXISTS budzet_potrosen (id INTEGER PRIMARY KEY AUTOINCREMENT, kategorija TEXT, stavka TEXT, kolicina INTEGER, cena_po_komadu REAL, ukupno REAL, dobavljac TEXT, datum DATETIME, korisnik TEXT)''')

    # Krpljenje
    try: c.execute("ALTER TABLE delovi ADD COLUMN dobavljac TEXT DEFAULT ''")
    except: pass
    try: c.execute("ALTER TABLE delovi ADD COLUMN cena REAL DEFAULT 0.0")
    except: pass
    try: c.execute("ALTER TABLE pm_linije ADD COLUMN ikona TEXT DEFAULT '🏭'")
    except: pass
    try: c.execute("ALTER TABLE pm_linije ADD COLUMN redosled INTEGER DEFAULT 0")
    except: pass
    try: c.execute("ALTER TABLE pm_masine ADD COLUMN redosled INTEGER DEFAULT 0")
    except: pass
    try: c.execute("ALTER TABLE pm_zadaci ADD COLUMN redosled INTEGER DEFAULT 0")
    except: pass

    # Početni podaci
    if c.execute("SELECT count(*) FROM korisnici").fetchone()[0] == 0:
        c.execute("INSERT INTO korisnici (username, password, uloga) VALUES ('admin', 'admin', 'admin')")
        c.execute("INSERT INTO korisnici (username, password, uloga) VALUES ('tehnicar', 'tehnicar', 'tehnicar')")
        
    if c.execute("SELECT count(*) FROM podesavanja WHERE kljuc='pm_pocetak'").fetchone()[0] == 0:
        c.execute("INSERT INTO podesavanja (kljuc, vrednost) VALUES ('pm_pocetak', ?)", (str(datetime.date.today()),))
        
    if c.execute("SELECT count(*) FROM podesavanja WHERE kljuc='reset_pin'").fetchone()[0] == 0:
        c.execute("INSERT INTO podesavanja (kljuc, vrednost) VALUES ('reset_pin', '300120')")
        
    conn.commit()
    conn.close()

class VirtualKeyboard(tk.Toplevel):
    def __init__(self, parent):
        super().__init__(parent)
        self.title("Virtuelna Tastatura")
        self.geometry("900x320")
        self.attributes('-topmost', True) 
        self.configure(bg="#18181b")
        self.protocol("WM_DELETE_WINDOW", self.hide)
        self.parent = parent
        self.is_shift = False
        self.keys_lower = [['1','2','3','4','5','6','7','8','9','0','Back'],['q','w','e','r','t','y','u','i','o','p','š','đ'],['a','s','d','f','g','h','j','k','l','č','ć'],['Shift','z','x','c','v','b','n','m','ž','Enter'],['Space']]
        self.keys_upper = [['!','"','#','$','%','&','/','(',')','=','Back'],['Q','W','E','R','T','Y','U','I','O','P','Š','Đ'],['A','S','D','F','G','H','J','K','L','Č','Ć'],['Shift','Z','X','C','V','B','N','M','Ž','Enter'],['Space']]
        self.create_widgets()
        self.withdraw() 

    def create_widgets(self):
        for widget in self.winfo_children(): widget.destroy()
        layout = self.keys_upper if self.is_shift else self.keys_lower
        for row in layout:
            f = tk.Frame(self, bg="#18181b"); f.pack(pady=4)
            for key in row:
                w = 4 if len(key) == 1 else (30 if key == 'Space' else 8)
                boja = "#3b82f6" if key in ['Shift', 'Enter', 'Back'] else "#3f3f46"
                tk.Button(f, text=key, font=("Segoe UI", 16, "bold"), bg=boja, fg="white", relief="flat", width=w, takefocus=0, command=lambda k=key: self.press(k)).pack(side="left", padx=4)

    def press(self, key):
        if key == 'Shift': 
            self.is_shift = not self.is_shift
            self.create_widgets()
            return
        target = getattr(self.parent, 'active_input', None)
        if not target: return
        try:
            if key == 'Back':
                idx = target.index(tk.INSERT)
                if idx > 0: target.delete(idx - 1)
            elif key == 'Space': target.insert(tk.INSERT, ' ')
            elif key == 'Enter': target.event_generate('<Return>')
            else: target.insert(tk.INSERT, key)
        except: pass 
        target.focus_set()

    def toggle(self):
        if self.winfo_ismapped(): self.withdraw()
        else:
            self.deiconify(); self.update_idletasks()
            self.geometry(f'+{(self.winfo_screenwidth()//2)-(self.winfo_width()//2)}+{self.winfo_screenheight()-self.winfo_height()-50}')

    def hide(self): self.withdraw()

class TouchDialog(tk.Toplevel):
    def __init__(self, parent, naslov, naziv_dela, max_kol, tip_transakcije):
        super().__init__(parent)
        self.title(naslov); self.geometry("600x750"); self.configure(bg="#27272a"); self.transient(parent)
        self.max_kol = max_kol; self.tip = tip_transakcije; self.result_kol = None; self.result_napomena = ""
        
        top_f = tk.Frame(self, bg="#27272a"); top_f.pack(pady=10, fill="x")
        ttk.Label(top_f, text=naziv_dela, font=("Segoe UI", 16, "bold"), background="#27272a", foreground="#0ea5e9", wraplength=550, justify="center").pack(pady=5)
        
        if self.tip in ["IZLAZ", "KORPA"]: ttk.Label(top_f, text=f"Dostupno: {max_kol}", font=("Segoe UI", 12), background="#27272a", foreground="#a1a1aa").pack()
        elif self.tip == "POVRAT": ttk.Label(top_f, text="Unesite količinu koju vraćate", font=("Segoe UI", 12, "bold"), background="#27272a", foreground="#f59e0b").pack()
        elif self.tip == "ULAZ": ttk.Label(top_f, text="Unesite nabavljenu količinu", font=("Segoe UI", 12), background="#27272a", foreground="#10b981").pack()
            
        self.lbl_display = tk.Label(self, text="0", font=("Segoe UI", 42, "bold"), bg="#18181b", fg="white", width=10, pady=10)
        self.lbl_display.pack(pady=15)
        
        f_napomena = tk.Frame(self, bg="#27272a"); f_napomena.pack(pady=5)
        ttk.Label(f_napomena, text="Napomena:", background="#27272a", foreground="white").pack(side="left")
        self.entry_nap = ttk.Entry(f_napomena, width=35); self.entry_nap.pack(side="left", padx=10)

        f_numpad = tk.Frame(self, bg="#27272a"); f_numpad.pack(pady=10)
        for r, row in enumerate([('7','8','9'), ('4','5','6'), ('1','2','3'), ('C','0','OK')]):
            for c, btn_text in enumerate(row):
                boja = "#0ea5e9" if btn_text == "OK" else ("#ef4444" if btn_text == "C" else "#3f3f46")
                tk.Button(f_numpad, text=btn_text, font=("Segoe UI", 24, "bold"), bg=boja, fg="white", width=5, relief="flat", command=lambda t=btn_text: self.pritisak(t)).grid(row=r, column=c, padx=5, pady=5)
                
        self.update_idletasks()
        self.geometry(f'+{(self.winfo_screenwidth()//2)-(self.winfo_width()//2)}+{(self.winfo_screenheight()//2)-(self.winfo_height()//2)}')
        self.wait_window()

    def pritisak(self, t):
        trenutno = self.lbl_display.cget("text")
        if t == "C": self.lbl_display.config(text="0")
        elif t == "OK":
            val = int(trenutno)
            if val <= 0: return messagebox.showerror("Greška", "Količina mora biti veća od 0!")
            if self.tip in ["IZLAZ", "KORPA"] and val > self.max_kol: return messagebox.showerror("Greška", f"Nedovoljno na stanju! (Max: {self.max_kol})")
            self.result_kol = val; self.result_napomena = self.entry_nap.get(); self.destroy()
        else: self.lbl_display.config(text=t if trenutno == "0" else trenutno + t)

        cols = 3 if len(stavke) > 4 else 2
        rows = (len(stavke) + cols - 1) // cols
        
        for i in range(cols): meni_grid.columnconfigure(i, weight=1, uniform="col")
        for i in range(rows): meni_grid.rowconfigure(i, weight=1, uniform="row")
        
        for idx, (ikona, naslov, opis, komanda, boja) in enumerate(stavke):
            pad_f = tk.Frame(meni_grid, bg=self.bg_main)
            pad_f.grid(row=idx // cols, column=idx % cols, sticky="nsew", padx=15, pady=15)
            
            btn = tk.Button(pad_f, bg=self.bg_card, relief="flat", activebackground=self.bg_card_hover, command=komanda)
            container = tk.Frame(btn, bg=self.bg_card); container.place(relx=0.5, rely=0.5, anchor="center")

            tk.Label(container, text=opis, font=("Segoe UI", 11), bg=self.bg_card, fg=self.text_muted).pack()

                if w != b: w.bind("<Button-1>", lambda e, cmd=komanda: cmd())
            
            bind_hover(btn); bind_hover(container)
            for child in container.winfo_children(): bind_hover(child)
            btn.pack(fill="both", expand=True)

        for r in conn.execute("SELECT part_number, naziv, stanje, min_stanje, cena, id FROM delovi WHERE kategorija=? AND (naziv LIKE ? OR part_number LIKE ?)", (self.trenutna_kategorija, search, search)): self.tree_d.insert("", "end", values=r)

    def osvezi_korpu(self):
        for i in self.tree_k.get_children(): self.tree_k.delete(i)
        for item in self.korpa: self.tree_k.insert("", "end", values=(item['naziv'], item['kol']))

    def akcija_kolicina(self, tip):
        sel = self.tree_d.focus()
        if not sel: return messagebox.showwarning("Greška", "Izaberite deo!")
        vals = self.tree_d.item(sel)['values']
        naziv = vals[1]; stanje = int(vals[2]); pid = vals[5]; cena = float(vals[4]) if vals[4] else 0.0
        
        d = TouchDialog(self, f"Unos: {tip}", naziv, 100000 if tip in ["ULAZ", "POVRAT"] else stanje, tip)
        if d.result_kol:
            if tip == "KORPA":
                self.korpa.append({'pid': pid, 'naziv': naziv, 'kol': d.result_kol, 'napomena': d.result_napomena})
                self.osvezi_korpu()
            else:
                conn = sqlite3.connect(DB_NAME)
                novo_stanje = stanje + d.result_kol if tip in ["ULAZ", "POVRAT"] else stanje - d.result_kol
                conn.execute("UPDATE delovi SET stanje=? WHERE id=?", (novo_stanje, pid))
                conn.execute("INSERT INTO transakcije (part_id, kolicina, tip, korisnik, datum, napomena) VALUES (?, ?, ?, ?, datetime('now', 'localtime'), ?)", (pid, d.result_kol, tip, self.trenutni_korisnik, d.result_napomena))
                
                if tip == "ULAZ":
                    ukupno = cena * d.result_kol
                    dob = conn.execute("SELECT dobavljac FROM delovi WHERE id=?", (pid,)).fetchone()[0] or "Nepoznato"
                    conn.execute("INSERT INTO budzet_potrosen (kategorija, stavka, kolicina, cena_po_komadu, ukupno, dobavljac, datum, korisnik) VALUES (?, ?, ?, ?, ?, ?, datetime('now', 'localtime'), ?)", (self.trenutna_kategorija, naziv, d.result_kol, cena, ukupno, dob, self.trenutni_korisnik))
                    messagebox.showinfo("Uspeh", f"Trošak budžeta: {ukupno:,.2f} RSD")
                conn.commit(); conn.close(); self.osvezi_tabelu_delova()

    def izdaj_korpu(self):
        if not self.korpa: return
        rn = f"RN-{datetime.datetime.now().strftime('%Y%m%d-%H%M%S')}"
        conn = sqlite3.connect(DB_NAME)
        conn.execute("INSERT INTO radni_nalozi (rn_broj, status, datum, korisnik) VALUES (?, 'Otvoren', datetime('now', 'localtime'), ?)", (rn, self.trenutni_korisnik))
        rn_id = conn.execute("SELECT last_insert_rowid()").fetchone()[0]
        
        for k in self.korpa:
            trenutno = conn.execute("SELECT stanje FROM delovi WHERE id=?", (k['pid'],)).fetchone()[0]
            conn.execute("UPDATE delovi SET stanje=? WHERE id=?", (trenutno - k['kol'], k['pid']))
            conn.execute("INSERT INTO transakcije (part_id, kolicina, tip, korisnik, datum, napomena) VALUES (?, ?, 'IZLAZ', ?, datetime('now', 'localtime'), ?)", (k['pid'], k['kol'], self.trenutni_korisnik, f"[{rn}] {k['napomena']}"))
            conn.execute("INSERT INTO rn_stavke (rn_id, part_id, kolicina) VALUES (?, ?, ?)", (rn_id, k['pid'], k['kol']))
        conn.commit(); conn.close(); self.korpa = []; self.osvezi_korpu(); self.osvezi_tabelu_delova()
        messagebox.showinfo("Uspeh", f"Radni nalog {rn} izdat!")

    def dodaj_mesece(self, datum, meseci):
        m = datum.month - 1 + meseci
        g = datum.year + m // 12
        m = m % 12 + 1
        dani_u_mesecu = [31, 29 if g % 4 == 0 and (g % 100 != 0 or g % 400 == 0) else 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]
        d = min(datum.day, dani_u_mesecu[m - 1])
        return datetime.date(g, m, d)

    def pm_get_status_and_date(self, z_id, tip_odrzavanja):
        conn = sqlite3.connect(DB_NAME)
        row = conn.execute("SELECT vrednost FROM podesavanja WHERE kljuc='pm_pocetak'").fetchone()
        start_str = row[0] if row else "2026-01-01"
        try: start_date = datetime.datetime.strptime(start_str, "%Y-%m-%d").date()
        except: start_date = datetime.date.today()
        
        last_ev = conn.execute("SELECT datum, korisnik FROM pm_evidencija WHERE zadatak_id=? ORDER BY datum DESC LIMIT 1", (z_id,)).fetchone()
        conn.close()

        datum_odrade = "-"
        korisnik = "-"
        
        if last_ev:
            datum_odrade = last_ev[0].split(".")[0]
            korisnik = last_ev[1]
            try: last_date = datetime.datetime.strptime(last_ev[0].split(" ")[0], "%Y-%m-%d").date()
            except: last_date = start_date
            
            if tip_odrzavanja == "Dnevno": next_date = last_date + datetime.timedelta(days=1)
            elif tip_odrzavanja == "Nedeljno": next_date = last_date + datetime.timedelta(days=7)
            elif tip_odrzavanja == "Mesečno": next_date = self.dodaj_mesece(last_date, 1)
            elif tip_odrzavanja == "Kvartalno": next_date = self.dodaj_mesece(last_date, 3)
            elif tip_odrzavanja == "Polugodišnje": next_date = self.dodaj_mesece(last_date, 6)
            elif tip_odrzavanja == "Godišnje": next_date = self.dodaj_mesece(last_date, 12)
            else: next_date = last_date + datetime.timedelta(days=365)
        else:
            next_date = start_date
            
        today = datetime.date.today()
        date_str = next_date.strftime("%Y-%m-%d")
        
        if today >= next_date: return "Kasni ⚠️", date_str, datum_odrade, korisnik
        else: return "U roku ☑", date_str, datum_odrade, korisnik

            idx1 = items.index(sel)
            idx2 = items.index(target)
            items[idx1], items[idx2] = items[idx2], items[idx1]
            
            conn = sqlite3.connect(DB_NAME)
            for i, item in enumerate(items):
                item_id = tv.item(item)['values'][0]
                conn.execute(f"UPDATE {tabela} SET redosled=? WHERE id=?", (i, item_id))
            conn.commit(); conn.close()
            
            target_id = tv.item(sel)['values'][0]
            if tabela == 'pm_linije': ucitaj_linije()
            elif tabela == 'pm_masine': ucitaj_masine(None)
            elif tabela == 'pm_zadaci': ucitaj_zadatke(None)
            
            for child in tv.get_children():
                if tv.item(child)['values'][0] == target_id:
                    tv.selection_set(child); tv.focus(child)
                    break

        def izmeni(tv, tabela):
            sel = tv.focus()
            if not sel: return messagebox.showwarning("Greška", "Izaberite stavku!")
            item_id = tv.item(sel)['values'][0]
            trenutni_naziv = tv.item(sel)['values'][2]
            
            w = tk.Toplevel(self); w.title("Izmena"); w.configure(bg=self.bg_panel)
            tk.Label(w, text="Novi Naziv/Opis:", bg=self.bg_panel, fg="white").pack(pady=5)
            e = ttk.Entry(w, width=40); e.insert(0, trenutni_naziv); e.pack(pady=5)
            
            cb_dodatak = None
            if tabela == 'pm_linije':
                ikona = tv.item(sel)['values'][3]
                tk.Label(w, text="Ikona:", bg=self.bg_panel, fg="white").pack(pady=5)
                cb_dodatak = ttk.Combobox(w, values=["🏭", "⚙️", "🌊", "⚡", "💨", "📦"], state="readonly")
                if ikona in cb_dodatak['values']: cb_dodatak.set(ikona)
                cb_dodatak.pack(pady=5)
            elif tabela == 'pm_zadaci':
                periodika = tv.item(sel)['values'][3]
                tk.Label(w, text="Periodika:", bg=self.bg_panel, fg="white").pack(pady=5)
                cb_dodatak = ttk.Combobox(w, values=["Dnevno", "Nedeljno", "Mesečno", "Kvartalno", "Polugodišnje", "Godišnje"], state="readonly")
                if periodika in cb_dodatak['values']: cb_dodatak.set(periodika)
                cb_dodatak.pack(pady=5)
                
            def spasi():
                if not e.get(): return
                conn = sqlite3.connect(DB_NAME)
                if tabela == 'pm_linije': conn.execute("UPDATE pm_linije SET naziv=?, ikona=? WHERE id=?", (e.get(), cb_dodatak.get(), item_id)); ucitaj_linije()
                elif tabela == 'pm_masine': conn.execute("UPDATE pm_masine SET naziv=? WHERE id=?", (e.get(), item_id)); ucitaj_masine(None)
                elif tabela == 'pm_zadaci': conn.execute("UPDATE pm_zadaci SET opis_zadatka=?, tip_odrzavanja=? WHERE id=?", (e.get(), cb_dodatak.get(), item_id)); ucitaj_zadatke(None)
                conn.commit(); conn.close(); w.destroy()
            ttk.Button(w, text="Sačuvaj izmene", style="Info.TButton", command=spasi).pack(pady=15)

        def unesi(tabela):
            w = tk.Toplevel(self); w.title("Dodaj"); w.configure(bg=self.bg_panel)
            tk.Label(w, text="Naziv/Opis:", bg=self.bg_panel, fg="white").pack(pady=5); e = ttk.Entry(w); e.pack(pady=5)
            cb_dodatak = None
            if tabela == 'pm_linije':
                tk.Label(w, text="Ikona linije:", bg=self.bg_panel, fg="white").pack(pady=5)
                cb_dodatak = ttk.Combobox(w, values=["🏭", "⚙️", "🌊", "⚡", "💨", "📦"], state="readonly")
                cb_dodatak.current(0); cb_dodatak.pack(pady=5)
            elif tabela == 'pm_zadaci':
                tk.Label(w, text="Periodika:", bg=self.bg_panel, fg="white").pack(pady=5)
                cb_dodatak = ttk.Combobox(w, values=["Dnevno", "Nedeljno", "Mesečno", "Kvartalno", "Polugodišnje", "Godišnje"], state="readonly")
                cb_dodatak.current(1); cb_dodatak.pack(pady=5)
            
            def spasi():
                if not e.get(): return
                conn = sqlite3.connect(DB_NAME)
                if tabela == 'pm_linije': conn.execute("INSERT INTO pm_linije (naziv, opis, ikona) VALUES (?, '', ?)", (e.get(), cb_dodatak.get())); ucitaj_linije()
                elif tabela == 'pm_masine': 
                    lid = tv_lin.item(tv_lin.focus())['values'][0]; conn.execute("INSERT INTO pm_masine (linija_id, naziv) VALUES (?, ?)", (lid, e.get())); ucitaj_masine(None)
                elif tabela == 'pm_zadaci': 
                    mid = tv_mas.item(tv_mas.focus())['values'][0]; conn.execute("INSERT INTO pm_zadaci (masina_id, opis_zadatka, tip_odrzavanja) VALUES (?, ?, ?)", (mid, e.get(), cb_dodatak.get())); ucitaj_zadatke(None)
                conn.commit(); conn.close(); w.destroy()
            ttk.Button(w, text="Sačuvaj", style="Success.TButton", command=spasi).pack(pady=15)

        def brisi(tv, tabela):
            sel = tv.focus()
            if not sel: return messagebox.showwarning("Greška", "Izaberite stavku!")
            dlg = ModernConfirm(self, "Brisanje PM Stavke", "Sigurno brišete stavku i sve njene zavisnosti iz PM baze?", "#ef4444")
            if dlg.result:
                conn = sqlite3.connect(DB_NAME)
                conn.execute(f"DELETE FROM {tabela} WHERE id=?", (tv.item(sel)['values'][0],))
                conn.commit(); conn.close()
                if tabela == 'pm_linije': ucitaj_linije()
                elif tabela == 'pm_masine': ucitaj_masine(None)
                elif tabela == 'pm_zadaci': ucitaj_zadatke(None)

        def kreiraj_kontrole(parent, tv, tabela, dodaj_cmd):
            f = tk.Frame(parent, bg=self.bg_panel); f.pack(fill="x", pady=2)
            tk.Button(f, text="⬆️", font=("Segoe UI", 12), bg="#3f3f46", fg="white", relief="flat", command=lambda: pomeri(tv, tabela, -1)).pack(side="left", fill="x", expand=True, padx=2)
            tk.Button(f, text="⬇️", font=("Segoe UI", 12), bg="#3f3f46", fg="white", relief="flat", command=lambda: pomeri(tv, tabela, 1)).pack(side="left", fill="x", expand=True, padx=2)
            ttk.Button(f, text="✏️ Izmeni", style="Info.TButton", command=lambda: izmeni(tv, tabela)).pack(side="left", fill="x", expand=True, padx=2)
            
            f2 = tk.Frame(parent, bg=self.bg_panel); f2.pack(fill="x", pady=2)
            ttk.Button(f2, text="➕ Dodaj", command=dodaj_cmd).pack(side="left", fill="x", expand=True, padx=2)
            ttk.Button(f2, text="🗑️ Obriši", style="Danger.TButton", command=lambda: brisi(tv, tabela)).pack(side="left", fill="x", expand=True, padx=2)

        kreiraj_kontrole(f_lin, tv_lin, 'pm_linije', lambda: unesi('pm_linije'))
        kreiraj_kontrole(f_mas, tv_mas, 'pm_masine', lambda: unesi('pm_masine'))
        kreiraj_kontrole(f_zad, tv_zad, 'pm_zadaci', lambda: unesi('pm_zadaci'))
        
        ucitaj_linije()
